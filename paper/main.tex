% The paper, anonymised for ISPASS 2027's double-blind review (paper skeleton
% wave, 29 Nov 2026). Structure follows docs/RESEARCH_PLAN.md, which wins over
% anything written here.
%
% Format, from the ISPASS 2027 submission page (fetched 29 Nov 2026): IEEE
% conference style, nine single-spaced double-column pages in 10-point type on
% 8.5x11 inch paper, with unlimited extra pages for references only; figures
% and appendices count within the nine; pages numbered; PDF only; references
% list all authors; author names and other hints of identity removed;
% substantial generative-AI use disclosed in the acknowledgments.
%
% Rules that bind this file and everything it inputs (plan Section 6):
%  - No numbers are typed into the text. Every number comes from a macro in
%    generated/numbers.tex (scripts/paper_figures.py) or sits in a generated
%    table or figure.
%  - Every exploratory or post hoc number carries its label.
%  - Cite only papers read in full (rule 5).
%  - No model identifiers (rule 9). Plain language, short sentences, no em dashes.
%
% \fact{...} bullets are drafting notes: facts from docs/paper/claim_status.md
% with their source. The author replaces them with prose; a submission build
% (paper/build.py --submission) refuses any \fact or \stub left.
%
% Build: python3 paper/build.py (see README.md in this directory).
\documentclass[conference]{IEEEtran}
\IEEEoverridecommandlockouts

\usepackage[T1]{fontenc}
\usepackage{amsmath}
\usepackage{graphicx}
\usepackage{booktabs}
\usepackage{xspace}
\usepackage{url}

\graphicspath{{figures/}}

% ---------------------------------------------------------------------------
% Generated numbers. A missing file stops the build; there is no fallback.
% ---------------------------------------------------------------------------
\providecommand{\NumbersFile}{generated/numbers.tex}
\IfFileExists{\NumbersFile}{%
  \input{\NumbersFile}%
}{%
  \errmessage{Generated numbers file '\NumbersFile' is missing. Run
    scripts/paper_figures.py. The paper never builds without it}%
  \csname @@end\endcsname
}

% Skeleton mode draws figures as labelled boxes (layout checks only).
\newif\ifskeleton
\providecommand{\SkeletonBuild}{0}
\ifnum\SkeletonBuild=1 \skeletontrue \fi

\newcommand{\paperfigure}[2][\columnwidth]{%
  \ifskeleton
    \fbox{\parbox[c][0.5\columnwidth][c]{0.95#1}{\centering
      \texttt{\detokenize{#2}}\\(skeleton placeholder)}}%
  \else
    \includegraphics[width=#1]{#2}%
  \fi}

% A visible marker for text not yet written.
\newcommand{\stub}[1]{\textbf{[Stub: #1]}}

% Drafting facts. \ShowFacts=0 hides them, to measure the layout without them.
\newif\iffacts
\providecommand{\ShowFacts}{1}
\ifnum\ShowFacts=1 \factstrue \fi
\newcommand{\src}[1]{\hfill{\scriptsize\texttt{\detokenize{#1}}}}
\newenvironment{facts}{%
  \iffacts\par\smallskip\noindent{\footnotesize\textit{Facts for this section
  (drafting notes, not prose):}}\footnotesize\raggedright\begin{itemize}\setlength{\itemsep}{0pt}%
  \else\setbox0\vbox\bgroup\begin{itemize}\fi}{%
  \iffacts\end{itemize}\normalsize\else\end{itemize}\egroup\fi}
\newcommand{\fact}[1]{\item #1}

% Records whether anything is cited, so the bibliography is typeset only then.
\newif\ifanycite
\let\papercite\cite
\renewcommand{\cite}{\global\anycitetrue\papercite}
\let\papernocite\nocite
\renewcommand{\nocite}{\global\anycitetrue\papernocite}
